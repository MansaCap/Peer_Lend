#Github Actions requirements

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-python@v4
        with:
          python-version: '3.1.9'
      - run: pip install -r requirements.txt
      - run: pytest
